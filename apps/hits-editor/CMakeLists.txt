cmake_minimum_required(VERSION 3.25)
project(HitsEditor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(../../cmake/hits_editor_core.cmake)
if(WIN32)
    add_executable(hits-editor WIN32 windows/main.cpp)
    target_link_libraries(hits-editor PRIVATE DMCRengine::HitsEditorCore comdlg32 gdi32 user32)
    target_compile_definitions(hits-editor PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MINGW)
        target_link_options(hits-editor PRIVATE -static -municode)
    endif()
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
    add_executable(hits-editor-viewport-tests ../../tests/hits_viewport_tests.cpp)
    target_link_libraries(hits-editor-viewport-tests PRIVATE DMCRengine::HitsEditorCore)
    if(MSVC)
        target_compile_options(hits-editor-viewport-tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(hits-editor-viewport-tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME hits-editor-viewport COMMAND hits-editor-viewport-tests)
endif()
